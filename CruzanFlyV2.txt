local function setupCell(object, name, parent, position, size, text, isLabel)
    object.Name = name
    object.Parent = parent
    object.BackgroundColor3 = WHITE
    object.BorderColor3 = DARK
    object.BorderSizePixel = 2
    object.Position = position
    object.Size = size
    object.Font = Enum.Font.SourceSans
    object.Text = text
    object.TextColor3 = TEXT
    object.TextSize = 14
    object.TextWrapped = true
    object.ZIndex = 3
    if isLabel then
        object.TextScaled = true
    end
end

-- A small dark backing below each cell makes the flat buttons look raised.
local shadows = {}
local function addRaisedEffect(object)
    local shadow = Instance.new("Frame")
    shadow.Name = object.Name .. "Shadow"
    shadow.Parent = object.Parent
    shadow.BackgroundColor3 = DARK
    shadow.BorderSizePixel = 0
    shadow.Position = object.Position + UDim2.new(0, 0, 0, 2)
    shadow.Size = object.Size
    shadow.ZIndex = 2
    shadow.Active = false
    shadow.Selectable = false
    shadows[object] = shadow

    -- Visible bottom bevel: it stays visible even where neighbouring cells touch.
    local bevel = Instance.new("Frame")
    bevel.Name = object.Name .. "BottomEdge"
    bevel.Parent = object
    bevel.AnchorPoint = Vector2.new(0, 1)
    bevel.Position = UDim2.new(0, 3, 1, 0)
    bevel.Size = UDim2.new(1, -6, 0, 2)
    bevel.BackgroundColor3 = DARK
    bevel.BorderSizePixel = 0
    bevel.ZIndex = object.ZIndex + 1
    bevel.Active = false
    bevel.Selectable = false
end

for _, object in ipairs({up, down, plus, mine, TextLabel, speed, noclipButton, onof, closebutton, mini, mini2}) do
    addRaisedEffect(object)
end

local function setVisible(object, isVisible)
    object.Visible = isVisible
    local shadow = shadows[object]
    if shadow then
        shadow.Visible = isVisible
    end
end

local function setPosition(object, position)
    object.Position = position
    local shadow = shadows[object]
    if shadow then
        shadow.Position = position + UDim2.new(0, 0, 0, 2)
    end
end

    pcall(function()

            Text = message,
            Duration = 3

    end)
end

local speeds = 1
local flightEnabled = false
local noclipEnabled = false
local shuttingDown = false

local humanoidStates = {
    Enum.HumanoidStateType.Climbing,
    Enum.HumanoidStateType.FallingDown,
    Enum.HumanoidStateType.Flying,
    Enum.HumanoidStateType.Freefall,
    Enum.HumanoidStateType.GettingUp,
    Enum.HumanoidStateType.Jumping,
    Enum.HumanoidStateType.Landed,
    Enum.HumanoidStateType.Physics,
    Enum.HumanoidStateType.PlatformStanding,
    Enum.HumanoidStateType.Ragdoll,
    Enum.HumanoidStateType.Running,
    Enum.HumanoidStateType.RunningNoPhysics,
    Enum.HumanoidStateType.Seated,
    Enum.HumanoidStateType.StrafingNoPhysics,
    Enum.HumanoidStateType.Swimming
}

local flyConnection = nil
local activeBodyGyro = nil
local activeBodyVelocity = nil
local flyCharacter = nil
local flyHumanoid = nil
local flyAnimate = nil
local savedPlatformStand = nil
local savedAnimateDisabled = nil
local savedStateEnabled = {}
local pausedTracks = {}

local function stopFly()
    flightEnabled = false

    if flyConnection then
        flyConnection:Disconnect()
        flyConnection = nil
    end

    if activeBodyGyro then
        pcall(function() activeBodyGyro:Destroy() end)
        activeBodyGyro = nil
    end
    if activeBodyVelocity then
        pcall(function() activeBodyVelocity:Destroy() end)
        activeBodyVelocity = nil
    end

    if flyHumanoid and flyHumanoid.Parent then
        for state, wasEnabled in pairs(savedStateEnabled) do
            pcall(function()
                flyHumanoid:SetStateEnabled(state, wasEnabled)
            end)
        end
        if savedPlatformStand ~= nil then
            pcall(function()
                flyHumanoid.PlatformStand = savedPlatformStand
            end)
        end
        if flyHumanoid.Health > 0 then
            pcall(function()
                flyHumanoid:ChangeState(Enum.HumanoidStateType.RunningNoPhysics)
            end)
        end
    end

    if flyAnimate and flyAnimate.Parent and savedAnimateDisabled ~= nil then
        pcall(function()
            flyAnimate.Disabled = savedAnimateDisabled
        end)
    end

    for _, track in ipairs(pausedTracks) do
        pcall(function()
            if track.IsPlaying then
                track:AdjustSpeed(1)
            end
        end)
    end

    savedStateEnabled = {}
    pausedTracks = {}
    flyCharacter = nil
    flyHumanoid = nil
    flyAnimate = nil
    savedPlatformStand = nil
    savedAnimateDisabled = nil
    onof.Text = "Fly"
end

local function startFly()
    if flightEnabled then
        return true
    end

    local character = player.Character
    local humanoid = character and character:FindFirstChildOfClass("Humanoid")
    local bodyPart = character and (character:FindFirstChild("Torso")
        or character:FindFirstChild("UpperTorso")
        or character:FindFirstChild("HumanoidRootPart"))

    if not character or not humanoid or not bodyPart or humanoid.Health <= 0 then
        notify("Cruzan", "Character is not ready yet")
        return false
    end

    flightEnabled = true
    flyCharacter = character
    flyHumanoid = humanoid
    flyAnimate = character:FindFirstChild("Animate")
    savedPlatformStand = humanoid.PlatformStand
    savedAnimateDisabled = flyAnimate and flyAnimate.Disabled or false
    savedStateEnabled = {}
    pausedTracks = {}

    for _, state in ipairs(humanoidStates) do
        local wasEnabled = true
        pcall(function()
            wasEnabled = humanoid:GetStateEnabled(state)
            humanoid:SetStateEnabled(state, false)
        end)
        savedStateEnabled[state] = wasEnabled
    end

    if flyAnimate then
        flyAnimate.Disabled = true
    end
    for _, track in ipairs(humanoid:GetPlayingAnimationTracks()) do
        table.insert(pausedTracks, track)
        pcall(function() track:AdjustSpeed(0) end)
    end

    humanoid.PlatformStand = true

    activeBodyGyro = Instance.new("BodyGyro")
    activeBodyGyro.Name = "CruzanFlyGyro"
    activeBodyGyro.P = 9e4
    activeBodyGyro.MaxTorque = Vector3.new(9e9, 9e9, 9e9)
    activeBodyGyro.CFrame = bodyPart.CFrame
    activeBodyGyro.Parent = bodyPart

    activeBodyVelocity = Instance.new("BodyVelocity")
    activeBodyVelocity.Name = "CruzanFlyVelocity"
    activeBodyVelocity.Velocity = Vector3.new(0, 0.1, 0)
    activeBodyVelocity.MaxForce = Vector3.new(9e9, 9e9, 9e9)
    activeBodyVelocity.Parent = bodyPart

    onof.Text = "Fly ON"

    flyConnection = RunService.Heartbeat:Connect(function(deltaTime)
        if not flightEnabled then
            return
        end
        if not flyCharacter or not flyCharacter.Parent or not flyHumanoid
            or not flyHumanoid.Parent or flyHumanoid.Health <= 0 then
            stopFly()
            return
        end

        local camera = workspace.CurrentCamera
        if activeBodyGyro and activeBodyGyro.Parent and camera then
            activeBodyGyro.CFrame = camera.CFrame
        end

        local moveDirection = flyHumanoid.MoveDirection
        if moveDirection.Magnitude > 0 then
            -- Matches the original multi-loop speed setting, but avoids duplicate loops.
            flyCharacter:TranslateBy(moveDirection * speeds * deltaTime * 60)
        end
    end)

    return true
end

local originalCollision = {}
local noclipConnection = nil

local function stopNoclip()
    noclipEnabled = false
    if noclipConnection then
        noclipConnection:Disconnect()
        noclipConnection = nil
    end

    for part, oldCanCollide in pairs(originalCollision) do
        if part and part.Parent and oldCanCollide ~= nil then
            pcall(function()
                part.CanCollide = oldCanCollide
            end)
        end
    end
    originalCollision = {}
    noclipButton.Text = "NoClip"
end

local function startNoclip()
    if noclipEnabled then
        return
    end

    noclipEnabled = true
    noclipButton.Text = "Clip"
    originalCollision = {}

    -- Uses Infinite Yield's Stepped approach, while saving each part's true prior state.
    noclipConnection = RunService.Stepped:Connect(function()
        if not noclipEnabled then
            return
        end
        local character = player.Character
        if not character then
            return
        end
        for _, part in ipairs(character:GetDescendants()) do
            if part:IsA("BasePart") then
                if originalCollision[part] == nil then
                    originalCollision[part] = part.CanCollide
                end
                if part.CanCollide then
                    part.CanCollide = false
                end
            end
        end
    end)
end

onof.Activated:Connect(function()
    if flightEnabled then
        stopFly()
    else
        startFly()
    end
end)

noclipButton.Activated:Connect(function()
    if noclipEnabled then
        stopNoclip()
    else
        startNoclip()
    end
end)

plus.Activated:Connect(function()
    speeds = speeds + 1
    speed.Text = tostring(speeds)
end)

mine.Activated:Connect(function()
    if speeds <= 1 then
        speed.Text = "1"
        return
    end
    speeds = speeds - 1
    speed.Text = tostring(speeds)
end)

-- UP/DOWN movement follows the same speed number as horizontal flight.
local holdingUp = false
local holdingDown = false
local heldInputs = {}

local function setVerticalHeld(direction, isHeld)
    if direction == "up" then
        holdingUp = isHeld
    elseif direction == "down" then
        holdingDown = isHeld
    end
end

local function bindHoldButton(button, direction)
    button.InputBegan:Connect(function(input)
        if input.UserInputType == Enum.UserInputType.MouseButton1
            or input.UserInputType == Enum.UserInputType.Touch then
            heldInputs[input] = direction
            setVerticalHeld(direction, true)
        end
    end)

    button.InputEnded:Connect(function(input)
        local heldDirection = heldInputs[input]
        if heldDirection then
            setVerticalHeld(heldDirection, false)
            heldInputs[input] = nil
        end
    end)
end

bindHoldButton(up, "up")
bindHoldButton(down, "down")

local inputEndedConnection = UserInputService.InputEnded:Connect(function(input)
    local heldDirection = heldInputs[input]
    if heldDirection then
        setVerticalHeld(heldDirection, false)
        heldInputs[input] = nil
    end
end)

local verticalConnection = RunService.Heartbeat:Connect(function(deltaTime)
    local direction = (holdingUp and 1 or 0) - (holdingDown and 1 or 0)
    if direction == 0 then
        return
    end
    local character = player.Character
    local root = character and character:FindFirstChild("HumanoidRootPart")
    if root then
        root.CFrame = root.CFrame + Vector3.new(0, direction * speeds * deltaTime * 60, 0)
    end
end)

local function minimizeGui()
    for _, object in ipairs({up, down, onof, noclipButton, TextLabel, plus, speed, mine, mini}) do
        setVisible(object, false)
    end
    setVisible(mini2, true)
    Frame.BackgroundTransparency = 1
    setPosition(closebutton, UDim2.new(0, 0, -1, 57))
end

local function restoreGui()
    for _, object in ipairs({up, down, onof, noclipButton, TextLabel, plus, speed, mine, mini}) do
        setVisible(object, true)
    end
    setVisible(mini2, false)
    Frame.BackgroundTransparency = 0
    setPosition(closebutton, UDim2.new(0, 0, -1, 27))
end

mini.Activated:Connect(minimizeGui)
mini2.Activated:Connect(restoreGui)

local characterAddedConnection = player.CharacterAdded:Connect(function()
    -- Avoid leaving either effect active across a respawn.
    stopFly()
    stopNoclip()
    holdingUp = false
    holdingDown = false
    heldInputs = {}
end)

local function cleanupAll()
    if shuttingDown then
        return
    end
    shuttingDown = true
    stopFly()
    stopNoclip()
    holdingUp = false
    holdingDown = false

    if verticalConnection then
        verticalConnection:Disconnect()
        verticalConnection = nil
    end
    if inputEndedConnection then
        inputEndedConnection:Disconnect()
    end
    if characterAddedConnection then
        characterAddedConnection:Disconnect()
    end
end

closebutton.Activated:Connect(function()
    cleanupAll()
    main:Destroy()
end)
