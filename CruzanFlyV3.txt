-- Each tile gets a complete dark offset backing, so the bottom/right edge is a real
-- external edge, not a partial line drawn inside the white button.
local buttonDepths = {}

local function updateButtonDepth(button)
    local depth = buttonDepths[button]
    if not depth then return end
    depth.Position = UDim2.new(
        button.Position.X.Scale, button.Position.X.Offset + 2,
        button.Position.Y.Scale, button.Position.Y.Offset + 3
    )
    depth.Size = button.Size
    depth.Visible = button.Visible
end

local function addRaisedDepth(button)
    local depth = Instance.new("Frame")
    depth.Name = button.Name .. "_Depth"
    depth.Parent = Frame
    depth.BackgroundColor3 = DARK
    depth.BorderSizePixel = 0
    depth.ZIndex = 2
    depth.Active = false
    buttonDepths[button] = depth
    updateButtonDepth(button)
end

for _, button in ipairs({up, down, plus, mine, noclipButton, onof, TextLabel, speed, closebutton, mini, mini2}) do
    addRaisedDepth(button)
end

local function setControlVisible(control, visible)
    control.Visible = visible
    local depth = buttonDepths[control]
    if depth then depth.Visible = visible end
end

local function stopWalkingWorkers()
    tpwalking = false
    walkingToken = walkingToken + 1
end

local function startWalkingWorkers()
    stopWalkingWorkers()
    tpwalking = true
    local token = walkingToken

    for _ = 1, speeds do
        spawn(function()
            local character = speaker.Character
            local humanoid = character and character:FindFirstChildWhichIsA("Humanoid")
            while tpwalking and token == walkingToken and nowe and character and humanoid and humanoid.Parent do
                RunService.Heartbeat:Wait()
                if tpwalking and token == walkingToken and nowe and humanoid.MoveDirection.Magnitude > 0 then
                    character:TranslateBy(humanoid.MoveDirection)
                end
            end
        end)
    end
end

local function stopVerticalMovement()
    verticalDirection = 0
    if verticalConnection then
        verticalConnection:Disconnect()
        verticalConnection = nil
    end
end

local function startVerticalMovement(direction)
    if not nowe then
        stopVerticalMovement()
        return
    end

    stopVerticalMovement()
    verticalDirection = direction
    verticalConnection = RunService.Heartbeat:Connect(function(deltaTime)
        if not nowe or verticalDirection == 0 then
            stopVerticalMovement()
            return
        end

        local character = speaker.Character
        local root = character and character:FindFirstChild("HumanoidRootPart")
        if not root then return end

        -- Same baseline as the original 1-stud-per-frame lift, multiplied by the
        -- selected speed. This only runs while Fly is enabled and the button is held.
        local verticalStep = verticalDirection * speeds * 60 * deltaTime
        root.CFrame = root.CFrame + Vector3.new(0, verticalStep, 0)
    end)
end

local function restoreAnimationSpeeds()
    for track, oldSpeed in pairs(savedAnimationSpeeds) do
        pcall(function()
            if track then track:AdjustSpeed(oldSpeed) end
        end)
    end
    savedAnimationSpeeds = {}
end

local function restoreCharacterAfterFlight(character)
    character = character or speaker.Character
    if not character then return end

    local humanoid = character:FindFirstChildWhichIsA("Humanoid")
    if humanoid then
        for _, state in ipairs(humanoidStatesToRestore) do
            pcall(function() humanoid:SetStateEnabled(state, true) end)
        end
        pcall(function() humanoid.PlatformStand = false end)
        pcall(function() humanoid:ChangeState(Enum.HumanoidStateType.RunningNoPhysics) end)
    end

    local animate = character:FindFirstChild("Animate")
    if animate then
        pcall(function() animate.Disabled = false end)
    end
    restoreAnimationSpeeds()
end

local function stopFlight()
    nowe = false
    flightToken = flightToken + 1
    stopWalkingWorkers()
    stopVerticalMovement()
    restoreCharacterAfterFlight(speaker.Character)
    onof.Text = "Fly"
end

local function setNoClip(enabled)
    noclipEnabled = enabled

    if noclipConnection then
        noclipConnection:Disconnect()
        noclipConnection = nil
    end

    if enabled then
        noclipParts = {}
        noclipButton.Text = "Clip"
        -- Adapted from the attached Infinite Yield noclip method.
        noclipConnection = RunService.Stepped:Connect(function()
            local character = speaker.Character
            if not noclipEnabled or not character then return end

            for _, part in ipairs(character:GetDescendants()) do
                if part:IsA("BasePart") then
                    if noclipParts[part] == nil then
                        noclipParts[part] = part.CanCollide
                    end
                    if part.CanCollide then
                        part.CanCollide = false
                    end
                end
            end
        end)
    else
        noclipButton.Text = "NoClip"
        for part, originalCanCollide in pairs(noclipParts) do
            if typeof(part) == "Instance" and part:IsA("BasePart") and part.Parent then
                pcall(function() part.CanCollide = originalCanCollide end)
            end
        end
        noclipParts = {}
    end
end
