local speeds = 1
local nowe = false
local tpwalking = false
local walkingToken = 0
local flightToken = 0
local verticalDirection = 0
local verticalConnection = nil
local noclipEnabled = false

local savedHumanoidStates = {}

local humanoidStatesToRestore = {
    Enum.HumanoidStateType.Climbing,
    Enum.HumanoidStateType.FallingDown,
    Enum.HumanoidStateType.Flying,
    Enum.HumanoidStateType.Freefall,
    Enum.HumanoidStateType.GettingUp,
    Enum.HumanoidStateType.Jumping,
    Enum.HumanoidStateType.Landed,
    Enum.HumanoidStateType.Physics,
    Enum.HumanoidStateType.PlatformStanding,
    Enum.HumanoidStateType.Ragdoll,
    Enum.HumanoidStateType.Running,
    Enum.HumanoidStateType.RunningNoPhysics,
    Enum.HumanoidStateType.Seated,
    Enum.HumanoidStateType.StrafingNoPhysics,
    Enum.HumanoidStateType.Swimming
}

local function stopWalkingWorkers()
    tpwalking = false
    walkingToken = walkingToken + 1
end

local function startWalkingWorkers()
    stopWalkingWorkers()
    tpwalking = true
    local token = walkingToken

    for _ = 1, speeds do
        spawn(function()
            local character = speaker.Character
            local humanoid = character and character:FindFirstChildWhichIsA("Humanoid")
            while tpwalking and token == walkingToken and nowe and character and humanoid and humanoid.Parent do
                RunService.Heartbeat:Wait()
                if tpwalking and token == walkingToken and nowe and humanoid.MoveDirection.Magnitude > 0 then
                    character:TranslateBy(humanoid.MoveDirection)
                end
            end
        end)
    end
end

local function stopVerticalMovement()
    verticalDirection = 0
    if verticalConnection then
        verticalConnection:Disconnect()
        verticalConnection = nil
    end
end

local function startVerticalMovement(direction)
    if not nowe then
        stopVerticalMovement()
        return
    end

    stopVerticalMovement()
    verticalDirection = direction
    verticalConnection = RunService.Heartbeat:Connect(function(deltaTime)
        if not nowe or verticalDirection == 0 then
            stopVerticalMovement()
            return
        end

        local character = speaker.Character
        local root = character and character:FindFirstChild("HumanoidRootPart")
        if not root then return end

        -- Same baseline as the original 1-stud-per-frame lift, multiplied by the
        -- selected speed. This only runs while Fly is enabled and the button is held.
        local verticalStep = verticalDirection * speeds * 60 * deltaTime
        root.CFrame = root.CFrame + Vector3.new(0, verticalStep, 0)
    end)
end

local function restoreAnimationSpeeds()
    for track, oldSpeed in pairs(savedAnimationSpeeds) do
        pcall(function()
            if track then track:AdjustSpeed(oldSpeed) end
        end)
    end
    savedAnimationSpeeds = {}
end

local function destroyFlightMovers()
    if activeBodyGyro then
        pcall(function() activeBodyGyro:Destroy() end)
        activeBodyGyro = nil
    end
    if activeBodyVelocity then
        pcall(function() activeBodyVelocity:Destroy() end)
        activeBodyVelocity = nil
    end
end

local function restoreCharacterAfterFlight(character)
    character = character or flightCharacter or speaker.Character
    if not character then
        restoreAnimationSpeeds()
        savedHumanoidStates = {}
        savedPlatformStand = nil
        savedAnimateDisabled = nil
        flightCharacter = nil
        return
    end

    local restoringFlightCharacter = (character == flightCharacter)
    local humanoid = character:FindFirstChildWhichIsA("Humanoid")
    if humanoid then
        for _, state in ipairs(humanoidStatesToRestore) do
            local wasEnabled = restoringFlightCharacter and savedHumanoidStates[state]
            if wasEnabled == nil then wasEnabled = true end
            pcall(function() humanoid:SetStateEnabled(state, wasEnabled) end)
        end

        local previousPlatformStand = restoringFlightCharacter and savedPlatformStand
        if previousPlatformStand == nil then previousPlatformStand = false end
        pcall(function() humanoid.PlatformStand = previousPlatformStand end)
        if not previousPlatformStand and humanoid.Health > 0 then
            -- Running is a normal controllable state; RunningNoPhysics can leave
            -- the character's jump state machine stuck after flight is disabled.
            pcall(function() humanoid:ChangeState(Enum.HumanoidStateType.Running) end)
        end
    end

    local animate = character:FindFirstChild("Animate")
    if animate then
        local previousAnimateDisabled = restoringFlightCharacter and savedAnimateDisabled
        if previousAnimateDisabled == nil then previousAnimateDisabled = false end
        pcall(function() animate.Disabled = previousAnimateDisabled end)
    end

    local root = character:FindFirstChild("HumanoidRootPart")
    if restoringFlightCharacter and root then
        pcall(function() root.AssemblyLinearVelocity = Vector3.zero end)
        pcall(function() root.AssemblyAngularVelocity = Vector3.zero end)
    end

    restoreAnimationSpeeds()
    if restoringFlightCharacter then
        savedHumanoidStates = {}
        savedPlatformStand = nil
        savedAnimateDisabled = nil
        flightCharacter = nil
    end
end

local function stopFlight()
    local wasFlying = nowe or flightCharacter ~= nil
    nowe = false
    flightToken = flightToken + 1
    stopWalkingWorkers()
    stopVerticalMovement()
    destroyFlightMovers()
    if wasFlying then
        restoreCharacterAfterFlight(flightCharacter or speaker.Character)
    end
    onof.Text = "Fly"
end

local function setNoClip(enabled)
    noclipEnabled = enabled

    if noclipConnection then
        noclipConnection:Disconnect()
        noclipConnection = nil
    end

    if enabled then
        noclipParts = {}
        noclipButton.Text = "Clip"
        -- Adapted from the attached Infinite Yield noclip method.
        noclipConnection = RunService.Stepped:Connect(function()
            local character = speaker.Character
            if not noclipEnabled or not character then return end

            for _, part in ipairs(character:GetDescendants()) do
                if part:IsA("BasePart") then
                    if noclipParts[part] == nil then
                        noclipParts[part] = part.CanCollide
                    end
                    if part.CanCollide then
                        part.CanCollide = false
                    end
                end
            end
        end)
    else
        noclipButton.Text = "NoClip"
        for part, originalCanCollide in pairs(noclipParts) do
            if typeof(part) == "Instance" and part:IsA("BasePart") and part.Parent then
                pcall(function() part.CanCollide = originalCanCollide end)
            end
        end
        noclipParts = {}
    end
end

-- Smooth touch/mouse dragging. Drag from Cruzan, the speed tile, or the
-- minimize controls so movement does not interfere with the UP/DOWN buttons.
local isDragging = false
local activeDragInput = nil
local dragStartPosition = nil
local frameStartPosition = nil

local function beginDrag(input)
    if input.UserInputType ~= Enum.UserInputType.MouseButton1
        and input.UserInputType ~= Enum.UserInputType.Touch then
        return
    end
    isDragging = true
    activeDragInput = input
    dragStartPosition = input.Position
    frameStartPosition = Frame.Position
end

local function addDragHandle(handle)
    handle.Active = true
    handle.InputBegan:Connect(beginDrag)
end

for _, handle in ipairs({TextLabel, speed, mini, mini2}) do
    addDragHandle(handle)
end

dragInputChangedConnection = UserInputService.InputChanged:Connect(function(input)
    if not isDragging or not activeDragInput then return end
    local isTouchDrag = activeDragInput.UserInputType == Enum.UserInputType.Touch
    if isTouchDrag then
        if input ~= activeDragInput then return end
    elseif input.UserInputType ~= Enum.UserInputType.MouseMovement then
        return
    end

    local delta = input.Position - dragStartPosition
    Frame.Position = UDim2.new(
        frameStartPosition.X.Scale, frameStartPosition.X.Offset + delta.X,
        frameStartPosition.Y.Scale, frameStartPosition.Y.Offset + delta.Y
    )
end)

dragInputEndedConnection = UserInputService.InputEnded:Connect(function(input)
    if not activeDragInput then return end
    if (activeDragInput.UserInputType == Enum.UserInputType.Touch and input == activeDragInput)
        or (activeDragInput.UserInputType == Enum.UserInputType.MouseButton1
            and input.UserInputType == Enum.UserInputType.MouseButton1) then
        isDragging = false
        activeDragInput = nil
        dragStartPosition = nil
        frameStartPosition = nil
    end
end)

onof.Activated:Connect(function()
    if nowe then
        stopFlight()
        return
    end

    local character = speaker.Character
    local humanoid = character and character:FindFirstChildWhichIsA("Humanoid")
    if not character or not humanoid or humanoid.Health <= 0 then
        return
    end

    nowe = true
    flightToken = flightToken + 1
    local activeFlightToken = flightToken
    flightCharacter = character
    savedHumanoidStates = {}
    for _, state in ipairs(humanoidStatesToRestore) do
        local ok, wasEnabled = pcall(function() return humanoid:GetStateEnabled(state) end)
        savedHumanoidStates[state] = (ok and wasEnabled) or false
    end
    savedPlatformStand = humanoid.PlatformStand

    local animateScript = character:FindFirstChild("Animate")
    savedAnimateDisabled = animateScript and animateScript.Disabled or false
    savedAnimationSpeeds = {}

    startWalkingWorkers()
    if animateScript then animateScript.Disabled = true end
    for _, track in ipairs(humanoid:GetPlayingAnimationTracks()) do
        local ok, oldSpeed = pcall(function() return track.Speed end)
        savedAnimationSpeeds[track] = (ok and type(oldSpeed) == "number") and oldSpeed or 1
        pcall(function() track:AdjustSpeed(0) end)
    end

    for _, state in ipairs(humanoidStatesToRestore) do
        pcall(function() humanoid:SetStateEnabled(state, false) end)
    end
    humanoid.PlatformStand = true
    humanoid:ChangeState(Enum.HumanoidStateType.Swimming)

    local plr = speaker
    local torso
    if humanoid.RigType == Enum.HumanoidRigType.R6 then
        torso = character:FindFirstChild("Torso")
    else
        torso = character:FindFirstChild("UpperTorso")
    end
    if not torso then
        stopFlight()
        return
    end

    local ctrl = {f = 0, b = 0, l = 0, r = 0}
    local lastctrl = {f = 0, b = 0, l = 0, r = 0}
    local maxspeed = 50
    local flightSpeed = 0

    local bg = Instance.new("BodyGyro")
    activeBodyGyro = bg
    bg.Parent = torso
    bg.P = 9e4
    bg.maxTorque = Vector3.new(9e9, 9e9, 9e9)
    bg.cframe = torso.CFrame

    local bv = Instance.new("BodyVelocity")
    activeBodyVelocity = bv
    bv.Parent = torso
    bv.velocity = Vector3.new(0, 0.1, 0)
    bv.maxForce = Vector3.new(9e9, 9e9, 9e9)

    while nowe and activeFlightToken == flightToken and character.Parent
        and humanoid.Parent and humanoid.Health > 0 do
        if humanoid.RigType == Enum.HumanoidRigType.R6 then
            RunService.RenderStepped:Wait()
        else
            task.wait()
        end

        if not (nowe and activeFlightToken == flightToken and character.Parent and humanoid.Parent and humanoid.Health > 0) then
            break
        end

        if ctrl.l + ctrl.r ~= 0 or ctrl.f + ctrl.b ~= 0 then
            flightSpeed = flightSpeed + 0.5 + (flightSpeed / maxspeed)
            if flightSpeed > maxspeed then flightSpeed = maxspeed end
        elseif flightSpeed ~= 0 then
            flightSpeed = math.max(0, flightSpeed - 1)
        end

        if ctrl.l + ctrl.r ~= 0 or ctrl.f + ctrl.b ~= 0 then
            bv.velocity = ((workspace.CurrentCamera.CoordinateFrame.lookVector * (ctrl.f + ctrl.b))
                + ((workspace.CurrentCamera.CoordinateFrame * CFrame.new(ctrl.l + ctrl.r, (ctrl.f + ctrl.b) * 0.2, 0).p)
                - workspace.CurrentCamera.CoordinateFrame.p)) * flightSpeed
            lastctrl = {f = ctrl.f, b = ctrl.b, l = ctrl.l, r = ctrl.r}
        elseif ctrl.l + ctrl.r == 0 and ctrl.f + ctrl.b == 0 and flightSpeed ~= 0 then
            bv.velocity = ((workspace.CurrentCamera.CoordinateFrame.lookVector * (lastctrl.f + lastctrl.b))
                + ((workspace.CurrentCamera.CoordinateFrame * CFrame.new(lastctrl.l + lastctrl.r, (lastctrl.f + lastctrl.b) * 0.2, 0).p)
                - workspace.CurrentCamera.CoordinateFrame.p)) * flightSpeed
        else
            bv.velocity = Vector3.new(0, 0, 0)
        end

        bg.cframe = workspace.CurrentCamera.CoordinateFrame
            * CFrame.Angles(-math.rad((ctrl.f + ctrl.b) * 50 * flightSpeed / maxspeed), 0, 0)
    end

    ctrl = {f = 0, b = 0, l = 0, r = 0}
    lastctrl = {f = 0, b = 0, l = 0, r = 0}
    flightSpeed = 0
    pcall(function() bg:Destroy() end)
    pcall(function() bv:Destroy() end)
    if activeBodyGyro == bg then activeBodyGyro = nil end
    if activeBodyVelocity == bv then activeBodyVelocity = nil end

    if activeFlightToken == flightToken then
        stopFlight()
    end
end)

closebutton.Activated:Connect(function()
    -- Shut down each feature independently before destroying the GUI.
    stopFlight()
    setNoClip(false)
    if characterAddedConnection then
        characterAddedConnection:Disconnect()
        characterAddedConnection = nil
    end
    if dragInputChangedConnection then
        dragInputChangedConnection:Disconnect()
        dragInputChangedConnection = nil
    end
    if dragInputEndedConnection then
        dragInputEndedConnection:Disconnect()
        dragInputEndedConnection = nil
    end
    main:Destroy()
end)
